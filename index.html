-- ============================================================ -- AgriScan Zambia - Create scans table -- Run this in your Supabase SQL Editor -- ============================================================ -- 1. Create the scans table CREATE TABLE scans ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, plant TEXT NOT NULL, disease TEXT NOT NULL, confidence INTEGER NOT NULL CHECK (confidence >= 0 AND confidence <= 100), recommendation TEXT, image_data TEXT, timestamp BIGINT NOT NULL, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 2. Create indexes for better performance CREATE INDEX idx_scans_plant ON scans(plant); CREATE INDEX idx_scans_disease ON scans(disease); CREATE INDEX idx_scans_timestamp ON scans(timestamp DESC); CREATE INDEX idx_scans_created_at ON scans(created_at DESC); -- 3. Enable Row Level Security (RLS) ALTER TABLE scans ENABLE ROW LEVEL SECURITY; -- 4. Create RLS policies (allow all operations for simplicity) -- For production, you may want to restrict these based on user authentication DROP POLICY IF EXISTS "Allow all operations" ON scans; CREATE POLICY "Allow all operations" ON scans FOR ALL USING (true) WITH CHECK (true); -- 5. Enable Realtime for the scans table -- This allows the app to receive live updates ALTER TABLE scans REPLICA IDENTITY FULL; ALTER PUBLICATION supabase_realtime ADD TABLE scans; -- 6. Create a stats view for easy reporting CREATE OR REPLACE VIEW scan_stats AS SELECT COUNT(*) AS total_scans, COUNT(CASE WHEN disease = 'Healthy' THEN 1 END) AS healthy_count, COUNT(CASE WHEN disease != 'Healthy' THEN 1 END) AS infection_count, ROUND(AVG(confidence), 1) AS avg_confidence FROM scans; -- 7.
